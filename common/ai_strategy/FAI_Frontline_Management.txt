# HER generic frontline management
#
# This layer is a fallback for countries without dedicated operational personalities.
# GER/SOV/ENG/USA/JAP/ITA/FRA/CHI are deliberately excluded: their country files
# contain phase-specific war plans that must not be overwritten by global rush/careful logic.

ai_front_management_stop = {
	enable = {
		has_war = yes
		NOT = { original_tag = GER }
		NOT = { original_tag = SOV }
		NOT = { original_tag = ENG }
		NOT = { original_tag = USA }
		NOT = { original_tag = JAP }
		NOT = { original_tag = ITA }
		NOT = { original_tag = FRA }
		NOT = { original_tag = CHI }
		has_manpower < 100000
		OR = {
			has_equipment = { ammunition_equipment < 1 }
			enemies_strength_ratio > 2.0
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_control
		country_trigger = { has_war_with = FROM }
		priority = 1000
		ordertype = front
		execution_type = careful
		execute_order = yes
		manual_attack = no
	}
}

# Tactical helpers are also fallback-only. Major war plans deliberately contain pauses
# and defensive phases; a global 5000/10000 rush rule would silently defeat those phases.
end_pocket = {
	enable = {
		has_war = yes
		NOT = { original_tag = GER }
		NOT = { original_tag = SOV }
		NOT = { original_tag = ENG }
		NOT = { original_tag = USA }
		NOT = { original_tag = JAP }
		NOT = { original_tag = ITA }
		NOT = { original_tag = FRA }
		NOT = { original_tag = CHI }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_control
		state_trigger = {
			OR = {
				AND = {
					is_controlled_by = FROM.FROM
					NOT = { is_fully_controlled_by = FROM.FROM }
				}
				controller = { has_war_with = FROM.FROM }
			}
			all_neighbor_state = {
				controller = { NOT = { has_war_with = FROM.FROM } }
			}
		}
		ratio = 0.0
		priority = 5000
		ordertype = front
		execution_type = rush
		execute_order = yes
		manual_attack = yes
	}
	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			OR = {
				AND = {
					is_controlled_by = FROM.FROM
					NOT = { is_fully_controlled_by = FROM.FROM }
				}
				controller = { has_war_with = FROM.FROM }
			}
			all_neighbor_state = {
				controller = { NOT = { has_war_with = FROM.FROM } }
			}
		}
		value = 40
	}
}

forced_check_omega_lul = {
	enable = { has_war = yes }
	abort_when_not_enabled = yes
	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			is_controlled_by = FROM.FROM
			NOT = {
				any_province_building_level = {
					province = { all_provinces = yes }
					building = supply_node
					level > 0
				}
			}
		}
		value = -5
	}
	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			OR = { is_capital = yes any_neighbor_state = { is_capital = yes } }
		}
		value = 15
	}
	ai_strategy = {
		type = front_unit_request
		state_trigger = {
			is_capital = yes
			owner = { tag = FROM.FROM }
		}
		value = 50
	}
	ai_strategy = {
		type = invasion_unit_request
		state_trigger = {
			NOT = { has_state_flag = recent_naval_invasion_flag }
			check_variable = { state_strategic_value < 3 }
		}
		value = -5
	}
	ai_strategy = {
		type = invasion_unit_request
		state_trigger = {
			NOT = { is_core_of = FROM.FROM }
			NOT = { has_state_flag = recent_naval_invasion_flag }
			check_variable = { state_strategic_value < 8 }
		}
