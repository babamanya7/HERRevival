only_civ_construction = {
	enable = {
		NOT = { original_tag = GER }
		NOT = { original_tag = SOV }
		NOT = { original_tag = ENG }
		NOT = { original_tag = USA }
		NOT = { original_tag = JAP }
		NOT = { original_tag = ITA }
		NOT = { original_tag = FRA }
		NOT = { original_tag = CHI }
		OR = {
			has_idea = undisturbed_isolation
			has_idea = isolation
			has_idea = civilian_economy
		}
		NOT = { has_war = yes }
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = building_target
		id = industrial_complex
		value = 120
	}
}

# Final division-template personality belongs to common/ai_templates + country strategies.
# The old global block gave every template category +100 and erased country differentiation.

civil_war_no_garrison = {
	enable = {
		has_civil_war = yes
		NOT = {
			any_enemy_country = { has_civil_war = no }
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = garrison value = -80 }
}

# At low tension reduce peacetime garrison demand, but never zero it out globally.

	enable = {
		threat < 0.15

	}
	abort_when_not_enabled = yes
	ai_strategy = { type = garrison value = -25 }
}
