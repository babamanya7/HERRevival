# HER Japanese AI strategy
#
# Design target after Hands-off Test #1:
# - win the mobile/expansion phase in China, then settle into an AI-vs-AI historical stalemate;
# - never abandon the Chinese coast while turning south;
# - execute the Southern Operation sequentially instead of splitting invasion capacity;
# - Philippines -> Malaya -> East Indies -> outer perimeter/New Guinea;
# - keep patrol/dominance and invasion-support naval goals active throughout the sequence;
# - avoid wasting the army in Australia/India before the historical perimeter is secured.

# Aggressive expansion is a late-1930s phase for AI-vs-AI. Against a human China,
# Japan remains unconstrained and can continue normal offensive behavior.

	allowed = { original_tag = JAP }
	enable = {
		has_war_with = CHI
		CHI = { has_capitulated = no }
		NOT = { has_war_with = USA }
		OR = {
			date < 1940.1.1
			CHI = { is_ai = no }
		}
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = front_unit_request tag = CHI value = 125 }
	ai_strategy = { type = area_priority id = north_china value = 190 }
	ai_strategy = { type = area_priority id = central_china value = 170 }
	ai_strategy = { type = area_priority id = south_china value = 90 }
	ai_strategy = { type = area_priority id = china_coast value = 150 }
	ai_strategy = { type = front_armor_score id = CHI value = -20 }

# AI-only historical freeze. Japan keeps a real holding army and protects the coast,
# but stops burning its stockpile in endless broad-front attacks after the main expansion.
JAP_china_historical_stalemate = {
	allowed = { original_tag = JAP }
	enable = {
		is_ai = yes
		has_war_with = CHI
		CHI = { is_ai = yes has_capitulated = no }
		date > 1939.12.31
	}
	abort_when_not_enabled = yes
	ai_strategy = { type = front_unit_request tag = CHI value = 80 }
	ai_strategy = { type = area_priority id = north_china value = 180 }
	ai_strategy = { type = area_priority id = central_china value = 150 }
	ai_strategy = { type = area_priority id = china_coast value = 220 }
	ai_strategy = { type = area_priority id = south_china value = 60 }
	ai_strategy = { type = front_control tag = CHI ratio = 0.65 priority = 120 ordertype = front execution_type = careful execute_order = no manual_attack = no }
}
