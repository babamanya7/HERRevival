#
# Design target:
# - slow peacetime buildup followed by overwhelming wartime mobilization;
# - carrier/screen-heavy two-ocean navy;
# - Europe-first strategic pressure without abandoning the Pacific;
# - active Pacific patrol, island defence and amphibious operations against Japan;
# - large fighter/naval-air arm and sufficient merchant shipping for global war.

# Reverse helper retained for alt-history democratic support behavior.
USA_protect_against_extremism = {
	reversed = yes
	enable_reverse = { USA = { has_government = democratic } }
	enable = {
		NOT = { tag = USA }
		NOT = { has_government = communism }
		NOT = { has_government = fascism }
		OR = {
			capital_scope = { is_on_continent = europe }
			AND = { num_of_civilian_factories > 10 is_subject = no }
		}
		any_of_scopes = {
			array = enemies
			NOT = { tag = USA }
			OR = { has_government = communism has_government = fascism }
		}
	}
	abort = {
		OR = {
			has_war = no
			has_government = communism
			has_government = fascism
			USA = { NOT = { has_government = democratic } }
			USA = { has_war_with = ROOT }
		}
	}
	ai_strategy = { type = support id = "USA" value = 100 }
}
