# Nationalist China: combat divisions stay at the HER 30-width standard.
# China economises through fewer expensive supports, not by fielding under-width frontline formations.

infantry_CHI = {
	available_for = { CHI }
	role = infantry

	upgrade_prio = { factor = 70 }

	CHI_infantry_early = {
		upgrade_prio = {
			factor = 40
			modifier = {
				factor = 0
				has_tech = support_weapons
			}
		}
		reinforce_prio = 2

		# Cheap but full-sized 30-width mass infantry for the opening China war.
		target_template = {
			support = {
				engineer = 1
				artillery = 1
			}
			regiments = {
				infantry = 15
			}
		}

		replace_at_match = 0.55
		replace_with = CHI_infantry_war
		target_min_match = 0.5
	}

	CHI_infantry_war = {
		enable = {
			has_tech = support_weapons
			has_tech = gw_artillery
		}
		upgrade_prio = { factor = 70 }
		reinforce_prio = 2

		# As artillery production improves, keep width and replace manpower with firepower.
		target_template = {
			support = {
				engineer = 1
				artillery = 1
				anti_air = 1
				anti_tank = 1
			}
			regiments = {
				heavy_infantry = 12
				artillery_brigade = 3
			}
		}

		replace_at_match = 0.7
		replace_with = CHI_infantry_late
		target_min_match = 0.6
	}

	CHI_infantry_late = {
		enable = {
			has_tech = support_weapons
			has_tech = heavy_artillery_1
			has_tech = interwar_antiair
			has_tech = interwar_antitank
		}
		upgrade_prio = { factor = 90 }
		reinforce_prio = 2

		# Late-war elite/regular line: still 30 width, but now with the support density China can sustain.
		target_template = {
			support = {
				engineer = 1
				recon = 1
				artillery = 1
				anti_air = 1
				anti_tank = 1
				signal_company = 1
				field_hospital = 1
			}
			regiments = {
				heavy_infantry = 9
				artillery_brigade = 3
				heavy_artillery_brigade = 1
			}
		}
	}
}
