### rubber part ###

on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					is_ai = no
				}
				set_variable = { num_of_rubber_storage = 0 }
				every_state = {
					limit = { is_controlled_by = PREV }
					PREV = { add_to_variable = { num_of_rubber_storage = PREV.non_damaged_building_level@rubber_storage } }
				}
				### Above, to count the number of rubber storages. ###
				
				set_variable = { rubber_speed = num_of_rubber_storage }
				multiply_variable = { rubber_speed = 500 }
				
				### Above, to count the speed of rubber storages putting out. ###
			
				set_variable = { rubber_cap = num_of_rubber_storage }
				multiply_variable = { rubber_cap = 100000 }
				
				set_variable = { storage_size_modifier = modifier@storage_size_factor }
				add_to_variable = { storage_size_modifier = 1 }
				multiply_variable = { rubber_cap = storage_size_modifier }
				round_variable = rubber_cap
				
				### Above, to count the total cap of rubber storages. ###
				
				set_variable = { daily_left_rubber = resource@rubber }
				clamp_variable = {
					var = daily_left_rubber
					min = 0
					max = 50000
				}
				
				### Above, to record the rubber left everyday. ###
				
				set_variable = { left_rubber = 0 }
				set_variable = { rubber_storage_use = 0 }
				set_variable = { rubber_storage_withdrawal = 0 }
				
				### Above, to init. ###
				
			}
		}
	}
    on_daily = {
		effect = {
			if = {
				limit = {
					is_ai = no
				}
				set_variable = { num_of_rubber_storage = 0 }
				every_state = {
					limit = { is_controlled_by = ROOT }
					ROOT = { add_to_variable = { num_of_rubber_storage = PREV.non_damaged_building_level@rubber_storage } }
				}
				### Above, to count the number of rubber storages. ###
				
				set_variable = { rubber_speed = num_of_rubber_storage }
				multiply_variable = { rubber_speed = 500 }
				
				### Above, to count the speed of rubber storages putting out. ###
				
				set_variable = { rubber_cap = num_of_rubber_storage }
				multiply_variable = { rubber_cap = 100000 }
				
				set_variable = { storage_size_modifier = modifier@storage_size_factor }
				add_to_variable = { storage_size_modifier = 1 }
				multiply_variable = { rubber_cap = storage_size_modifier }
				round_variable = rubber_cap
				
				### Above, to count the total cap of rubber storages. ###
			}
			# Keep two quantities separate:
			# storage_use is the raw dynamic-modifier value;
			# storage_withdrawal is the real amount delivered and removed from stock.
			if = {
				limit = { is_ai = no }
				set_variable = { res_eff_size_modifier = 1 }
				add_to_variable = { res_eff_size_modifier = modifier@local_resources_factor }
				clamp_variable = {
					var = res_eff_size_modifier
					min = 0.01
					max = 100
				}

				# Reconstruct yesterday's real output after country resource modifiers.
				set_variable = { rubber_storage_previous_output = rubber_storage_use }
				multiply_variable = { rubber_storage_previous_output = res_eff_size_modifier }

				# Only genuine domestic/trade surplus enters storage.
				set_variable = { daily_left_rubber = resource@rubber }
				subtract_from_variable = { daily_left_rubber = rubber_storage_previous_output }
				clamp_variable = {
					var = daily_left_rubber
					min = 0
					max = 50000
				}

				if = {
					limit = { has_country_flag = rubber_storage_use }

					# Desired real withdrawal: old output minus the balance visible today.
					set_variable = { rubber_storage_withdrawal = rubber_storage_previous_output }
					subtract_from_variable = { rubber_storage_withdrawal = resource@rubber }
					clamp_variable = {
						var = rubber_storage_withdrawal
						min = 0
						max = rubber_speed
					}

					if = {
						limit = { check_variable = { rubber_storage_withdrawal > 0 } }
						if = {
							limit = {
								OR = {
									check_variable = { left_rubber > rubber_storage_withdrawal }
									check_variable = { left_rubber = rubber_storage_withdrawal }
								}
							}
							subtract_from_variable = { left_rubber = rubber_storage_withdrawal }
						}
						else = {
							set_variable = { rubber_storage_withdrawal = left_rubber }
							set_variable = { left_rubber = 0 }
							if = {
								limit = { NOT = { has_country_flag = rubber_storage_empty } }
								country_event = { id = her_storage.1 }
							}
							set_country_flag = rubber_storage_empty
						}
					}

					# Counteract local_resources_factor for the technical modifier only.
					set_variable = { rubber_storage_use = rubber_storage_withdrawal }
					divide_variable = { rubber_storage_use = res_eff_size_modifier }
				}
				else = {
					set_variable = { rubber_storage_use = 0 }
					set_variable = { rubber_storage_withdrawal = 0 }
				}

				add_to_variable = { left_rubber = daily_left_rubber }
				
				### Above, to store the rubber ###
				
				clamp_variable = {
					var = left_rubber
					min = 0
					max = rubber_cap
				}
				
				### Above, to clamp the storage ###
				
				if = {
					limit = { 
						ROOT = { 
							OR = {
								check_variable = { left_rubber > rubber_speed }
								check_variable = { left_rubber = rubber_speed } 
							}
						} 
					}
					ROOT = { clr_country_flag = rubber_storage_empty }
				}
				
				### Above, Let the storages work again, when the storage become enough. ###
				
				### Notice: Because in 1936.1.1 there is no storage, so it should take out first, then put in. ###
			}
		}
    }
}

					var = daily_left_chromium
					min = 0
					max = 50000
				}
				set_variable = { left_chromium = 0 }
				set_variable = { chromium_storage_use = 0 }
				set_variable = { chromium_storage_withdrawal = 0 }
			}
		}
	}
    on_daily = {
		effect = {
			if = {
				limit = {
					is_ai = no
				}
				set_variable = { num_of_chromium_storage = 0 }
				every_state = {
					limit = { is_controlled_by = ROOT }
					ROOT = { add_to_variable = { num_of_chromium_storage = PREV.non_damaged_building_level@chromium_storage } }
				}
				### Above, to count the number of chromium storages. ###
				
				set_variable = { chromium_speed = num_of_chromium_storage }
				multiply_variable = { chromium_speed = 600 }
				
				### Above, to count the speed of chromium storages putting out. ###
				
				set_variable = { chromium_cap = num_of_chromium_storage }
				multiply_variable = { chromium_cap = 75000 }
				
				set_variable = { storage_size_modifier = modifier@storage_size_factor }
				add_to_variable = { storage_size_modifier = 1 }
				multiply_variable = { chromium_cap = storage_size_modifier }
				round_variable = chromium_cap
				
				### Above, to count the total cap of chromium storages. ###
			}
			# Keep two quantities separate:
			# storage_use is the raw dynamic-modifier value;
			# storage_withdrawal is the real amount delivered and removed from stock.
			if = {
				limit = { is_ai = no }
				set_variable = { res_eff_size_modifier = 1 }
				add_to_variable = { res_eff_size_modifier = modifier@local_resources_factor }
				clamp_variable = {
					var = res_eff_size_modifier
					min = 0.01
					max = 100
				}

				# Reconstruct yesterday's real output after country resource modifiers.
				set_variable = { chromium_storage_previous_output = chromium_storage_use }
				multiply_variable = { chromium_storage_previous_output = res_eff_size_modifier }

				# Only genuine domestic/trade surplus enters storage.
				set_variable = { daily_left_chromium = resource@chromium }
				subtract_from_variable = { daily_left_chromium = chromium_storage_previous_output }
				clamp_variable = {
					var = daily_left_chromium
					min = 0
					max = 50000
				}

				if = {
					limit = { has_country_flag = chromium_storage_use }

					# Desired real withdrawal: old output minus the balance visible today.
					set_variable = { chromium_storage_withdrawal = chromium_storage_previous_output }
					subtract_from_variable = { chromium_storage_withdrawal = resource@chromium }
					clamp_variable = {
						var = chromium_storage_withdrawal
						min = 0
						max = chromium_speed
					}

					if = {
						limit = { check_variable = { chromium_storage_withdrawal > 0 } }
						if = {
							limit = {
								OR = {
									check_variable = { left_chromium > chromium_storage_withdrawal }
									check_variable = { left_chromium = chromium_storage_withdrawal }
								}
							}
							subtract_from_variable = { left_chromium = chromium_storage_withdrawal }
						}
						else = {
							set_variable = { chromium_storage_withdrawal = left_chromium }
							set_variable = { left_chromium = 0 }
							if = {
								limit = { NOT = { has_country_flag = chromium_storage_empty } }
								country_event = { id = her_storage.5 }
							}
							set_country_flag = chromium_storage_empty
						}
					}

					# Counteract local_resources_factor for the technical modifier only.
					set_variable = { chromium_storage_use = chromium_storage_withdrawal }
					divide_variable = { chromium_storage_use = res_eff_size_modifier }
				}
				else = {
					set_variable = { chromium_storage_use = 0 }
					set_variable = { chromium_storage_withdrawal = 0 }
				}

				add_to_variable = { left_chromium = daily_left_chromium }
				
				### Above, to store the chromium ###
				
				clamp_variable = {
					var = left_chromium
					min = 0
					max = chromium_cap
				}
				
				### Above, to clamp the storage ###
				
				if = {
					limit = { 
						ROOT = { 
							OR = {
								check_variable = { left_chromium > chromium_speed }
								check_variable = { left_chromium = chromium_speed } 
							}
						} 
					}
					ROOT = { clr_country_flag = chromium_storage_empty }
				}
				
				### Above, Let the storages work again, when the storage become enough. ###
				
				### Notice: Because in 1936.1.1 there is no storage, so it should take out first, then put in. ###
			}
		}
    }
