### rubber part ###

on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					is_ai = no
				}
				set_variable = { num_of_rubber_storage = 0 }
				every_state = {
					limit = { is_controlled_by = PREV }
					PREV = { add_to_variable = { num_of_rubber_storage = PREV.non_damaged_building_level@rubber_storage } }
				}
				### Above, to count the number of rubber storages. ###
				
				set_variable = { rubber_speed = num_of_rubber_storage }
				multiply_variable = { rubber_speed = 500 }
				
				### Above, to count the speed of rubber storages putting out. ###
			
				set_variable = { rubber_cap = num_of_rubber_storage }
				multiply_variable = { rubber_cap = 100000 }
				
				set_variable = { storage_size_modifier = modifier@storage_size_factor }
				add_to_variable = { storage_size_modifier = 1 }
				multiply_variable = { rubber_cap = storage_size_modifier }
				round_variable = rubber_cap
				
				### Above, to count the total cap of rubber storages. ###
				
				set_variable = { daily_left_rubber = resource@rubber }
				clamp_variable = {
					var = daily_left_rubber
					min = 0
					max = 50000
				}
				
				### Above, to record the rubber left everyday. ###
				
				set_variable = { left_rubber = 0 }
				set_variable = { rubber_storage_use = 0 }
				
				### Above, to init. ###
				
			}
		}
	}
    on_daily = {
		effect = {
			if = {
				limit = {
					is_ai = no
				}
				set_variable = { num_of_rubber_storage = 0 }
				every_state = {
					limit = { is_controlled_by = ROOT }
					ROOT = { add_to_variable = { num_of_rubber_storage = PREV.non_damaged_building_level@rubber_storage } }
				}
				### Above, to count the number of rubber storages. ###
				
				set_variable = { rubber_speed = num_of_rubber_storage }
				multiply_variable = { rubber_speed = 500 }
				
				### Above, to count the speed of rubber storages putting out. ###
				
				set_variable = { rubber_cap = num_of_rubber_storage }
				multiply_variable = { rubber_cap = 100000 }
				
				set_variable = { storage_size_modifier = modifier@storage_size_factor }
				add_to_variable = { storage_size_modifier = 1 }
				multiply_variable = { rubber_cap = storage_size_modifier }
				round_variable = rubber_cap
				
				### Above, to count the total cap of rubber storages. ###
			}
			if = {
				limit = {
					is_ai = no
					ROOT = { has_country_flag = rubber_storage_use }
					ROOT = { check_variable = { resource@rubber < 0 } }
				}
				ROOT = {
					set_variable = { rubber_storage_use_temp = 0 }
					add_to_variable = { rubber_storage_use_temp = resource@rubber }
					multiply_variable = { rubber_storage_use_temp = -1 }
					clamp_variable = {
						var = rubber_storage_use_temp
						min = 0
						max = rubber_speed
					}
					add_to_variable = { rubber_storage_use = rubber_storage_use_temp }
					clamp_variable = {
						var = rubber_storage_use
						min = 0
						max = rubber_speed
					}
					set_variable = { res_eff_size_modifier = 1 }
					add_to_variable = { res_eff_size_modifier = modifier@local_resources_factor }
					divide_variable = { rubber_storage_use = res_eff_size_modifier }
					round_variable = rubber_storage_use
					### rubber_storage_use, This var will be used in dynamic modifier. ###
				}
			}
			if = {
				limit = {
					is_ai = no
					ROOT = { has_country_flag = rubber_storage_use }
					ROOT = { check_variable = { resource@rubber > 0 } }
				}
				set_variable = { rubber_storage_use_temp = 0 }
				add_to_variable = { rubber_storage_use_temp = resource@rubber }
				subtract_from_variable = { rubber_storage_use = rubber_storage_use_temp }
			}
			if = {
				limit = {
					is_ai = no
					ROOT = { has_country_flag = rubber_storage_use }
					ROOT = { check_variable = { rubber_storage_use > 0 } }
				}
				if = {
					limit = { ROOT = { check_variable = { left_rubber > rubber_speed } } }
					subtract_from_variable = { left_rubber = rubber_storage_use }
				}
				else = {
					ROOT = {
						if = { 
							limit = { ROOT = { NOT = { has_country_flag = rubber_storage_empty } } }
							set_country_flag = rubber_storage_empty
							set_variable = { rubber_storage_use = left_rubber }
							set_variable = { left_rubber = 0 }
						}
						
						### Above, If the left rubber is less than the max of the building that one day can putting out, just let the putting out number be the left. ###
						### VERY IMPORTANT!!! MAKE A FLAG to let the storages not to put out before its left more than the speed putting out !!! ###
						
						else = {
							set_variable = { rubber_storage_use = 0 }
						}
						### Above, Let it invaild ###
						
					}
				}
			}
			if = {
				limit = {
					is_ai = no
					ROOT = { NOT = { has_country_flag = rubber_storage_use } }
				}
				set_variable = { rubber_storage_use = 0 }
			}
			if = {
				limit = {
					is_ai = no
				}
				set_variable = { daily_left_rubber = resource@rubber }
				subtract_from_variable = { daily_left_rubber = rubber_storage_use }
				clamp_variable = {
					var = daily_left_rubber
					min = 0
					max = 50000
				}
				add_to_variable = { left_rubber = daily_left_rubber }
				
				### Above, to store the rubber ###
				
				clamp_variable = {
					var = left_rubber
					min = 0
					max = rubber_cap
				}
				
				### Above, to clamp the storage ###
				
				if = {
					limit = { 
						ROOT = { 
							OR = {
								check_variable = { left_rubber > rubber_speed }
								check_variable = { left_rubber = rubber_speed } 
							}
						} 
					}
					ROOT = { clr_country_flag = rubber_storage_empty }
				}
				
				### Above, Let the storages work again, when the storage become enough. ###
				
				### Notice: Because in 1936.1.1 there is no storage, so it should take out first, then put in. ###
			}
		}
    }
}

				limit = {
					is_ai = no
					ROOT = { has_country_flag = iron_storage_use }
					ROOT = { check_variable = { iron_storage_use > 0 } }
				}
				if = {
					limit = { ROOT = { check_variable = { left_iron > iron_speed } } }
					subtract_from_variable = { left_iron = iron_storage_use }
				}
				else = {
					ROOT = {
						if = { 
							limit = { ROOT = { NOT = { has_country_flag = iron_storage_empty } } }
							set_country_flag = iron_storage_empty
							set_variable = { iron_storage_use = left_iron }
							set_variable = { left_iron = 0 }
						}
						
						### Above, If the left iron is less than the max of the building that one day can putting out, just let the putting out number be the left. ###
						### VERY IMPORTANT!!! MAKE A FLAG to let the storages not to put out before its left more than the speed putting out !!! ###
						
						else = {
							set_variable = { iron_storage_use = 0 }
						}
						### Above, Let it invaild ###
						
					}
				}
			}
			if = {
				limit = {
					is_ai = no
					ROOT = { NOT = { has_country_flag = iron_storage_use } }
				}
				set_variable = { iron_storage_use = 0 }
			}
			if = {
				limit = {
					is_ai = no
				}
				set_variable = { daily_left_iron = resource@iron }
				subtract_from_variable = { daily_left_iron = iron_storage_use }
				clamp_variable = {
					var = daily_left_iron
					min = 0
					max = 50000
				}
				add_to_variable = { left_iron = daily_left_iron }
				
				### Above, to store the iron ###
				
				clamp_variable = {
					var = left_iron
					min = 0
					max = iron_cap
				}
				
				### Above, to clamp the storage ###
				
				if = {
					limit = { 
						ROOT = { 
							OR = {
								check_variable = { left_iron > iron_speed }
								check_variable = { left_iron = iron_speed } 
							}
						} 
					}
					ROOT = { clr_country_flag = iron_storage_empty }
				}
				
				### Above, Let the storages work again, when the storage become enough. ###
				
				### Notice: Because in 1936.1.1 there is no storage, so it should take out first, then put in. ###
			}
		}
