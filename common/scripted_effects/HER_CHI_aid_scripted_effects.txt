# The American commodity credit requires a warehouse away from the exposed mining regions.
# The flag also prevents repeated slot creation. Keeping this as a daily-safe effect migrates old saves.
HER_CHI_ensure_chongqing_export_depot = {
	if = {
		limit = {
			has_country_flag = CHI_usa_tungsten_credit_active
			NOT = { has_country_flag = CHI_usa_export_depot_established }
			605 = { is_controlled_by = ROOT }
		}
		605 = {
			add_extra_state_shared_building_slots = 1
			add_building_construction = { type = tungsten_storage level = 1 instant_build = yes }
		}
		set_country_flag = CHI_usa_export_depot_established
	}
}

HER_CHI_update_western_route_capacity = {
	set_variable = { CHI_sea_route_capacity = 0 }
	set_variable = { CHI_hanoi_route_capacity = 0 }
	set_variable = { CHI_burma_route_capacity = 0 }
	set_variable = { CHI_hump_route_capacity = 0 }
	set_variable = { CHI_ledo_route_capacity = 0 }
	set_variable = { CHI_western_surface_capacity = 0 }
	set_variable = { CHI_sea_import_capacity = 0 }
	set_variable = { CHI_sea_export_capacity = 0 }
	set_variable = { CHI_hanoi_import_capacity = 0 }
	set_variable = { CHI_hanoi_export_capacity = 0 }
	set_variable = { CHI_burma_import_capacity = 0 }
	set_variable = { CHI_burma_export_capacity = 0 }
	set_variable = { CHI_hump_import_capacity = 0 }
	set_variable = { CHI_ledo_import_capacity = 0 }
	set_variable = { CHI_ledo_export_capacity = 0 }
	clamp_variable = { var = CHI_burma_route_upgrade min = 0 max = 2500 }
	clamp_variable = { var = CHI_hump_route_upgrade min = 0 max = 1500 }
	clamp_variable = { var = CHI_ledo_route_upgrade min = 0 max = 2500 }

	if = { limit = { 592 = { CONTROLLER = { NOT = { has_war_with = ROOT } } } } add_to_variable = { CHI_sea_route_capacity = 1250 } }
	if = { limit = { 593 = { CONTROLLER = { NOT = { has_war_with = ROOT } } } } add_to_variable = { CHI_sea_route_capacity = 1250 } }
	if = { limit = { 595 = { CONTROLLER = { NOT = { has_war_with = ROOT } } } } add_to_variable = { CHI_sea_route_capacity = 1250 } }
	if = { limit = { 613 = { CONTROLLER = { NOT = { has_war_with = ROOT } } } } add_to_variable = { CHI_sea_route_capacity = 1250 } }
	if = {
		limit = { HER_CHI_hanoi_route_open = yes }
		set_variable = { CHI_hanoi_route_capacity = 2500 }
	}
	if = {
		limit = { HER_CHI_burma_route_open = yes }
		set_variable = { CHI_burma_route_capacity = 2500 }
		add_to_variable = { var = CHI_burma_route_capacity value = CHI_burma_route_upgrade }
	}
	if = {
		limit = { HER_CHI_hump_route_open = yes }
		set_variable = { CHI_hump_route_capacity = 1000 }
		add_to_variable = { var = CHI_hump_route_capacity value = CHI_hump_route_upgrade }
	}
	if = {
		limit = { HER_CHI_ledo_route_open = yes }
		set_variable = { CHI_ledo_route_capacity = 2500 }
		add_to_variable = { var = CHI_ledo_route_capacity value = CHI_ledo_route_upgrade }
	}

	add_to_variable = { var = CHI_western_surface_capacity value = CHI_sea_route_capacity }
	add_to_variable = { var = CHI_western_surface_capacity value = CHI_hanoi_route_capacity }
	add_to_variable = { var = CHI_western_surface_capacity value = CHI_burma_route_capacity }
	add_to_variable = { var = CHI_western_surface_capacity value = CHI_ledo_route_capacity }
	set_variable = { CHI_sea_import_capacity = CHI_sea_route_capacity }
	set_variable = { CHI_sea_export_capacity = CHI_sea_route_capacity }
	set_variable = { CHI_hanoi_import_capacity = CHI_hanoi_route_capacity }
	set_variable = { CHI_hanoi_export_capacity = CHI_hanoi_route_capacity }
	set_variable = { CHI_burma_import_capacity = CHI_burma_route_capacity }
	set_variable = { CHI_burma_export_capacity = CHI_burma_route_capacity }
	set_variable = { CHI_hump_import_capacity = CHI_hump_route_capacity }
	set_variable = { CHI_ledo_import_capacity = CHI_ledo_route_capacity }
	set_variable = { CHI_ledo_export_capacity = CHI_ledo_route_capacity }

	if = {
		limit = { has_country_flag = CHI_transport_priority_military }
		multiply_variable = { CHI_sea_import_capacity = 0.8 }
		multiply_variable = { CHI_sea_export_capacity = 0.2 }
		multiply_variable = { CHI_hanoi_import_capacity = 0.8 }
		multiply_variable = { CHI_hanoi_export_capacity = 0.2 }
		multiply_variable = { CHI_burma_import_capacity = 0.8 }
		multiply_variable = { CHI_burma_export_capacity = 0.2 }
		multiply_variable = { CHI_ledo_import_capacity = 0.8 }
		multiply_variable = { CHI_ledo_export_capacity = 0.2 }
	}
	else_if = {
		limit = { has_country_flag = CHI_transport_priority_credit }
		multiply_variable = { CHI_sea_import_capacity = 0.4 }
		multiply_variable = { CHI_sea_export_capacity = 0.6 }
		multiply_variable = { CHI_hanoi_import_capacity = 0.4 }
		multiply_variable = { CHI_hanoi_export_capacity = 0.6 }
		multiply_variable = { CHI_burma_import_capacity = 0.4 }
		multiply_variable = { CHI_burma_export_capacity = 0.6 }
		multiply_variable = { CHI_ledo_import_capacity = 0.4 }
		multiply_variable = { CHI_ledo_export_capacity = 0.6 }
	}
	else = {
		multiply_variable = { CHI_sea_import_capacity = 0.6 }
		multiply_variable = { CHI_sea_export_capacity = 0.4 }
		multiply_variable = { CHI_hanoi_import_capacity = 0.6 }
		multiply_variable = { CHI_hanoi_export_capacity = 0.4 }
		multiply_variable = { CHI_burma_import_capacity = 0.6 }
		multiply_variable = { CHI_burma_export_capacity = 0.4 }
		multiply_variable = { CHI_ledo_import_capacity = 0.6 }
		multiply_variable = { CHI_ledo_export_capacity = 0.4 }
	}
	set_variable = { CHI_sea_import_allocation = CHI_sea_import_capacity }
	set_variable = { CHI_sea_export_allocation = CHI_sea_export_capacity }
	set_variable = { CHI_hanoi_import_allocation = CHI_hanoi_import_capacity }
	set_variable = { CHI_hanoi_export_allocation = CHI_hanoi_export_capacity }
	set_variable = { CHI_burma_import_allocation = CHI_burma_import_capacity }
	set_variable = { CHI_burma_export_allocation = CHI_burma_export_capacity }
	set_variable = { CHI_hump_import_allocation = CHI_hump_import_capacity }
	set_variable = { CHI_ledo_import_allocation = CHI_ledo_import_capacity }
	set_variable = { CHI_ledo_export_allocation = CHI_ledo_export_capacity }

	set_variable = { CHI_western_import_capacity = 0 }
	add_to_variable = { var = CHI_western_import_capacity value = CHI_sea_import_capacity }
	add_to_variable = { var = CHI_western_import_capacity value = CHI_hanoi_import_capacity }
	add_to_variable = { var = CHI_western_import_capacity value = CHI_burma_import_capacity }
	add_to_variable = { var = CHI_western_import_capacity value = CHI_hump_import_capacity }
	add_to_variable = { var = CHI_western_import_capacity value = CHI_ledo_import_capacity }
	set_variable = { CHI_western_export_capacity = 0 }
	add_to_variable = { var = CHI_western_export_capacity value = CHI_sea_export_capacity }
	add_to_variable = { var = CHI_western_export_capacity value = CHI_hanoi_export_capacity }
	add_to_variable = { var = CHI_western_export_capacity value = CHI_burma_export_capacity }
	add_to_variable = { var = CHI_western_export_capacity value = CHI_ledo_export_capacity }
	# The Hump carries aid into China, but is not used to export bulk tungsten.
	set_variable = { CHI_western_route_capacity = CHI_western_surface_capacity }
	add_to_variable = { var = CHI_western_route_capacity value = CHI_hump_route_capacity }
	round_variable = CHI_western_route_capacity
	round_variable = CHI_western_import_capacity
	round_variable = CHI_western_export_capacity
	set_variable = { CHI_western_import_allocation = CHI_western_import_capacity }
	set_variable = { CHI_western_export_allocation = CHI_western_export_capacity }
}

# Select one physical route for the current packet. The caller sets
# CHI_current_cargo_weight before invoking the selector.
HER_CHI_select_import_route = {
	clr_country_flag = CHI_cargo_route_sea
	clr_country_flag = CHI_cargo_route_hanoi
	clr_country_flag = CHI_cargo_route_burma
	clr_country_flag = CHI_cargo_route_hump
	clr_country_flag = CHI_cargo_route_ledo
	if = {
		limit = { NOT = { check_variable = { CHI_sea_import_capacity < CHI_current_cargo_weight } } }
		subtract_from_variable = { var = CHI_sea_import_capacity value = CHI_current_cargo_weight }
		set_country_flag = CHI_cargo_route_sea
	}
	else_if = {
		limit = { NOT = { check_variable = { CHI_hanoi_import_capacity < CHI_current_cargo_weight } } }
		subtract_from_variable = { var = CHI_hanoi_import_capacity value = CHI_current_cargo_weight }
		set_country_flag = CHI_cargo_route_hanoi
	}
	else_if = {
		limit = { NOT = { check_variable = { CHI_burma_import_capacity < CHI_current_cargo_weight } } }
		subtract_from_variable = { var = CHI_burma_import_capacity value = CHI_current_cargo_weight }
		set_country_flag = CHI_cargo_route_burma
	}
	else_if = {
		limit = { NOT = { check_variable = { CHI_ledo_import_capacity < CHI_current_cargo_weight } } }
		subtract_from_variable = { var = CHI_ledo_import_capacity value = CHI_current_cargo_weight }
		set_country_flag = CHI_cargo_route_ledo
	}
	else_if = {
		limit = { NOT = { check_variable = { CHI_hump_import_capacity < CHI_current_cargo_weight } } }
		subtract_from_variable = { var = CHI_hump_import_capacity value = CHI_current_cargo_weight }
		set_country_flag = CHI_cargo_route_hump
	}
}

	set_variable = { CHI_current_cargo_weight = 250 }
	if = {
		limit = {
			HER_CHI_export_route_can_fit_current_cargo = yes
			check_variable = { CHI_usa_tungsten_debt > 999 }
			check_variable = { left_tungsten > 999 }
			USA = { exists = yes }
		}
		HER_CHI_select_export_route = yes
		subtract_from_variable = { CHI_western_export_capacity = 250 }
		subtract_from_variable = { CHI_usa_tungsten_debt = 1000 }
		subtract_from_variable = { left_tungsten = 1000 }
		add_to_variable = { CHI_tungsten_in_transit = 1000 }
		add_to_variable = { CHI_tungsten_exported_last_day = 1000 }
		HER_CHI_schedule_tungsten_arrival = yes
	}
	if = {
		limit = {
			has_country_flag = CHI_usa_tungsten_credit_active
			check_variable = { CHI_usa_tungsten_debt < 1 }
		}
		set_country_flag = CHI_usa_tungsten_debt_repaid
		clr_country_flag = CHI_usa_tungsten_credit_active
	}
}
