		set_unit_leader_flag = her_skip_terrain_on_add
		if = { limit = { check_variable = { her_winter_lvl > 0 } } meta_effect = { text = { remove_unit_leader_trait = her_winter_[X] } X = "[?her_winter_lvl]" } }
		add_to_variable = { her_winter_lvl = 1 }
		if = {
			limit = { check_variable = { her_winter_lvl = 10 } }
			add_unit_leader_trait = winter_specialist
		}
		else = { meta_effect = { text = { add_unit_leader_trait = her_winter_[X] } X = "[?her_winter_lvl]" } }
		clr_unit_leader_flag = her_skip_terrain_on_add
	}
	if = {
		limit = { OR = { check_variable = { num_units_in_combat > 0 } has_unit_leader_flag = recent_combat } }
		her_update_terrain_xp_sample = yes
		unit_leader_event = { id = her_terrain_traits.1 hours = 8 }
		unit_leader_event = { id = her_terrain_traits.2 hours = 16 }
	}
}
