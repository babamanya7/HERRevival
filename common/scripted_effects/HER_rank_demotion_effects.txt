		limit = { has_trait = gen_pk }

	}

		limit = { has_trait = tank_gen_fm }
		remove_unit_leader_trait = tank_gen_fm
		add_unit_leader_trait = tank_gen_arm
		demote_leader = yes
	}

		limit = { has_trait = tank_gen_arm }
		remove_unit_leader_trait = tank_gen_arm
		add_unit_leader_trait = tank_gen_pk

		limit = { has_trait = tank_gen_pk }
		remove_unit_leader_trait = tank_gen_pk
		add_unit_leader_trait = tank_gen_lt
	}
