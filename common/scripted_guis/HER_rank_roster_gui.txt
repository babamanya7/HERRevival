scripted_gui = {

	HER_rank_roster_launcher_gui = {
		window_name = "HER_rank_roster_launcher"
		parent_window_name = "countryofficercorpview"

		visible = {
			always = yes
		}

		effects = {
			open_rank_roster_click = {
				clear_array = HER_rank_roster_array
				every_unit_leader = {
					limit = {
						OR = {
							has_trait = gen_lt
							has_trait = gen_pk
							has_trait = gen_arm
							has_trait = gen_fm
							has_trait = tank_gen_lt
							has_trait = tank_gen_pk
							has_trait = tank_gen_arm
							has_trait = tank_gen_fm
						}
					}
					add_to_array = { ROOT.HER_rank_roster_array = THIS }
				}
				set_country_flag = HER_rank_roster_open
			}
		}
	}

	HER_rank_roster_window_gui = {
		window_name = "HER_rank_roster_window"
		parent_window_name = "countryofficercorpview"

		visible = {
			has_country_flag = HER_rank_roster_open
		}

		dynamic_lists = {
			rank_roster_grid = {
				array = HER_rank_roster_array
				change_scope = yes
				entry_container = "HER_rank_roster_entry"
			}
		}

		effects = {
			close_rank_roster_click = {
				clr_country_flag = HER_rank_roster_open
			}

			her_rank_demote_click = {
				HER_demote_unit_leader_rank = yes
			}
		}

		triggers = {
			her_rank_demote_click_enabled = {
				OR = {
					has_trait = gen_pk
					has_trait = gen_arm
					has_trait = gen_fm
					has_trait = tank_gen_pk
					has_trait = tank_gen_arm
					has_trait = tank_gen_fm
				}
			}
		}
	}
}
