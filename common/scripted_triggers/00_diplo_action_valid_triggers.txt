	OR = {
		AND = {
			tag = ENG
			FROM = {
				NOT = { is_playable_country = yes }
				has_war_with = GER
			}
		}
		AND = {
			tag = GER
			OR = {
				AND = {
					date > 1941.1.1
					FROM = { the_axis = yes }
				}
				AND = {
					FROM = { tag = ITA }
					date > 1940.7.1
				}
			}
		}
	}
}
