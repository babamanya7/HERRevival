# HER Naval Institute Rework
# Specialized torpedo development. Electric and oxygen torpedoes are alternative paths.

sp_naval_electric_torpedo = {
	specialization = specialization_naval
	project_tags = { sp_tag_submarine sp_tag_surface_ship }

	available = {
		FROM = {
			has_tech = magnetic_detonator
			NOT = { has_tech = oxygen_torpedo }
			NOT = { is_special_project_completed = sp:sp_naval_oxygen_torpedo }
		}
	}

	breakthrough_cost = {
		specialization_naval = 2
	}

	resource_cost = {
		resources = {
			steel = 40
			aluminium = 40
			copper = 40
		}
	}

	prototype_time = sp_time.prototype.medium
	complexity = sp_complexity.medium

	project_output = {
		country_effects = {
			custom_effect_tooltip = {
				localization_key = SP_UNLOCK_TECH
				TECH = electric_torpedo
			}
			custom_override_tooltip = {
				tooltip = sp_naval_electric_torpedo_output_tt
				set_technology = {
					electric_torpedo = 1
					popup = no
				}
			}
		}
	}

	generic_prototype_rewards = {
		sp_naval_generic_reward_scientist_xp_1
		sp_naval_generic_reward_scientist_xp_2
		sp_naval_generic_reward_navy_xp_1
		sp_naval_generic_reward_navy_xp_2
		sp_naval_generic_reward_major_progress_1
		sp_naval_generic_reward_major_progress_2
		sp_naval_generic_reward_test_failure_1
		sp_naval_generic_reward_test_failure_2
	}
}
