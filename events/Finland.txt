#Surrender when Soviet has taken too much land
country_event = {
	id = finland.7
	title = finland.7.t
	desc = finland.7.d
	picture = GFX_report_event_soviet_tanks_snow
	
	mean_time_to_happen = { days = 14 }
	
	trigger = {
		tag = FIN
		has_war_with = SOV
		OR = {
			146 = { NOT = { is_controlled_by = FIN } }
			surrender_progress > 0.21
			AND = {
				SOV = { is_ai = yes }
				date > 1940.3.1
			}
		}
		NOT = { has_global_flag = FIN_winter_war_peace_has_been }
		NOT = { is_in_faction_with = GER }
	}

	#Surrender and offer territory
	option = {
		name = finland.7.a
		ai_chance = {
			base = 8
		}
		SOV = {
			country_event = { id = finland.8 days = 1 }
		}
		effect_tooltip = {
			SOV = { 
				white_peace = {
					tag = FIN 
					message = FIN_agree_peace 
				} 
			}
			transfer_state = 146
			transfer_state = 147
		}
		hidden_effect = {
			set_country_flag = FIN_winter_War_lost
			set_global_flag = FIN_winter_war_peace_has_been
		}
	}
}

	id = finland.11

	is_triggered_only = yes

	#Accept surrender
	option = {
		name = finland.11.a
		every_country = {
			limit = {
				has_war_with = FIN
			}
			white_peace = FIN
		}
		transfer_state = 722
		transfer_state = 147
		if = {
			limit = {
				NOT = {
					SOV = {
						owns_state = 146
					}
				}
			}
			transfer_state = 146
		}
		GER = { 
			remove_from_faction = FIN 
		}
		GER = { 
			add_opinion_modifier = { target = FIN modifier = finland_surrenderd }
			reverse_add_opinion_modifier = { target = FIN modifier = finland_betrayed_us }
			add_opinion_modifier = { target = FIN modifier = finland_betrayed_us }
			reverse_add_opinion_modifier = { target = FIN modifier = finland_betrayed_us }
		}
		FIN = {
			add_popularity = {
				ideology = communism
				popularity = 0.3
			}
			add_ideas = {
				disarmed_nation
			}
			diplomatic_relation = { country = SOV relation = non_aggression_pact }
			diplomatic_relation = { country = ENG relation = non_aggression_pact }
			diplomatic_relation = { country = USA relation = non_aggression_pact }
			remove_state_claim = 213
			remove_state_claim = 215
			remove_state_claim = 216
			set_country_flag = finland_is_done

	}
}
