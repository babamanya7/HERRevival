# Historical cleanup helper: when AI USA is at war with Vichy, hand the two
# relevant states to USA. Keep this as a bounded map-state helper, not a combat buff.
country_event = {
	id = her_ai.5
	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = BRA
		USA = {
			is_ai = yes
			has_war_with = VIC
		}
	}

	immediate = {
		USA = { transfer_state = 461 }
		USA = { transfer_state = 462 }
	}

	option = {}
}
