	title = her_chi_aid.1.t
	desc = her_chi_aid.1.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	option = {
		name = her_chi_aid.1.a
		ai_chance = { factor = 100 }
		CHI = {
			clr_country_flag = CHI_usa_tungsten_credit_requested
			set_country_flag = CHI_usa_tungsten_credit_active
			add_to_variable = { CHI_usa_credit_balance = 500000 }
			add_to_variable = { CHI_usa_tungsten_debt = 1000000 }
			country_event = { id = her_chi_aid.2 days = 1 }
		}
	}

	option = {
		name = her_chi_aid.1.b
		ai_chance = { factor = 0 }

			set_country_flag = CHI_usa_tungsten_credit_refused
			country_event = { id = her_chi_aid.3 days = 1 }
		}
	}
}

	id = her_chi_aid.4
	title = her_chi_aid.4.t
	desc = her_chi_aid.4.d
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes
	option = {
		name = her_chi_aid.contract_accept
		trigger = {
			check_variable = { CHI_usa_credit_balance > 19999 }
			USA = { has_equipment = { infantry_equipment > 4999 } }
		}
		subtract_from_variable = { CHI_usa_credit_balance = 20000 }
		USA = {
			clr_country_flag = USA_CHI_rifle_offer_pending
			send_equipment = { target = BRA type = infantry_equipment amount = 5000 old_prioritised = yes }
		}
		BRA = { add_to_variable = { BRA_CHI_pending_infantry_equipment = 5000 } }
	}
	option = {
		name = her_chi_aid.contract_refuse
		USA = { clr_country_flag = USA_CHI_rifle_offer_pending }
	}
}
