# Route-dispatched packets spend seven days in transit. Arrivals update both a rolling
# seven-day total and a permanent cumulative total in Chinese country scope.
country_event = {
	id = her_chi_aid.120
	hidden = yes
	is_triggered_only = yes
	immediate = {
		send_equipment = { target = CHI type = infantry_equipment amount = 500 old_prioritised = yes }
		send_equipment = { target = CHI type = machinegun amount = 240 old_prioritised = yes }
		send_equipment = { target = CHI type = mortar amount = 80 old_prioritised = yes }
		send_equipment = { target = CHI type = antitank amount = 70 old_prioritised = yes }
		subtract_from_variable = { BRA_CHI_in_transit_infantry_equipment = 500 }
		subtract_from_variable = { BRA_CHI_in_transit_machinegun = 240 }
		subtract_from_variable = { BRA_CHI_in_transit_mortar = 80 }
		subtract_from_variable = { BRA_CHI_in_transit_antitank = 70 }
		CHI = {
			add_to_variable = { CHI_arrived_7d_infantry_equipment = 500 }
			add_to_variable = { CHI_arrived_7d_machinegun = 240 }
			add_to_variable = { CHI_arrived_7d_mortar = 80 }
			add_to_variable = { CHI_arrived_7d_antitank = 70 }
			add_to_variable = { CHI_arrived_total_infantry_equipment = 500 }
			add_to_variable = { CHI_arrived_total_machinegun = 240 }
			add_to_variable = { CHI_arrived_total_mortar = 80 }
			add_to_variable = { CHI_arrived_total_antitank = 70 }
		}
		country_event = { id = her_chi_aid.140 days = 7 }
	}
	option = { }
}

# Chinese tungsten uses the same physical seven-day transit as Western equipment.
country_event = {
	id = her_chi_aid.128
	hidden = yes
	is_triggered_only = yes
	immediate = {
		subtract_from_variable = { CHI_tungsten_in_transit = 1000 }
		clamp_variable = { var = CHI_tungsten_in_transit min = 0 max = 100000000 }
		add_to_variable = { CHI_tungsten_arrived_7d = 1000 }
		add_to_variable = { CHI_tungsten_arrived_total = 1000 }
		USA = { add_to_variable = { left_tungsten = 1000 } }
		country_event = { id = her_chi_aid.148 days = 7 }
	}
	option = { }
}
