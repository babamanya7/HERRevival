		hidden_effect = {

		name = wtt_japan.109.a #accept
		add_political_power = 50
		ai_chance = {
			base = 10
		}

		effect_tooltip = {
			JAP = { add_ideas = JAP_tripartite_pact }
		}
		event_target:ITA_fascism = {
			country_event = { id = wtt_japan.110 hours = 6 }
		}
	}
}

	id = wtt_japan.111

	desc = wtt_japan.111.desc

	is_triggered_only = yes
	
	option = {
		name = excellent
		add_political_power = 50
		JAP = { add_ideas = JAP_tripartite_pact }
		add_opinion_modifier = { target = GER modifier = tripartite_pact_trade }
		reverse_add_opinion_modifier = { target = GER modifier = tripartite_pact_trade }
		hidden_effect = {
			set_global_flag = tripartite_pact_signed
		}
		ai_chance = {
			base = 10
		}
		effect_tooltip = {
			event_target:JAP_initiator = {
				give_guarantee = event_target:ITA_fascism
				give_guarantee = event_target:GER_fascism
			}
			event_target:GER_fascism = {
				give_guarantee = event_target:ITA_fascism
				give_guarantee = event_target:JAP_initiator
			}
			event_target:ITA_fascism = {
				give_guarantee = event_target:JAP_initiator
				give_guarantee = event_target:GER_fascism
			}
		}
		hidden_effect = {
			every_country = {
				limit = {
					is_literally_china = yes
				}
				event_target:ITA_fascism = {
					add_ai_strategy = {
						type = alliance

						value = -500
					}
				}
				event_target:GER_fascism = {
					add_ai_strategy = {
						type = alliance
						id = PREV
						value = -500
					}
				}
			}
		}
	}
}
