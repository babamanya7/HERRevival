				USA = { controls_province = 13349 }
				USA = { controls_province = 13233 }
				USA = { controls_province = 7632 }
				USA = { controls_province = 9294 }
			}
			count_triggers = {
				amount = 3
				ENG = { controls_province = 13376 }
				ENG = { controls_province = 13494 }
				ENG = { controls_province = 13349 }
				ENG = { controls_province = 13233 }
				ENG = { controls_province = 7632 }
				ENG = { controls_province = 9294 }
			}
			count_triggers = {
				amount = 3
				FRA = { controls_province = 13376 }
				FRA = { controls_province = 13494 }
				FRA = { controls_province = 13349 }
				FRA = { controls_province = 13233 }
				FRA = { controls_province = 7632 }
				FRA = { controls_province = 9294 }
			}
		}
	}

	immediate = {
		set_global_flag = d_day_happened
		news_event = { id = afo_news.1 }
		GER = { country_event = { id = ger_armor.57 days = 5 } }
	}

	option = {}

}
