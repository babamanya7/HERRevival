	id = ss_recruitment_event.11
	title = ss_recruitment_event.11.t	
	desc = ss_recruitment_event.11.desc	#
	picture = GFX_report_event_germany_politician_handshake	

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			random_unit_leader = {
				limit = {
					skill > 3
					NOT = {
						has_trait = politically_connected
					}
				}
				save_event_target_as = GER_SS_general_resigns
			}
		}
	}

		custom_effect_tooltip = GER_Wehrmacht_attitude_5_tt

			base = 25
		}

			retire = yes
		}
		add_to_variable = { var = GER_Wehrmacht_attitude value = 5 }
	}
	option = { #placate
		name = ss_recruitment_event.11.b
		custom_effect_tooltip = GER_Wehrmacht_attitude_5_tt

			base = 75
			modifier = {
				factor = 0
