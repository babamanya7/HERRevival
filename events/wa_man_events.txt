add_namespace = man_armor

### Soviet notification of raids

country_event = {
	id = man_armor.801
	title = man_armor.801.t
	desc = man_armor.801.d

	trigger = {
		date > 1936.1.18
		tag = SOV
	}
	
	fire_only_once = yes

	option = {
		name = man_armor.801.a
		custom_effect_tooltip = MAN_raids_tt
	}
}

### Chita

country_event = {
	id = man_armor.802
	title = man_armor.802.t
	desc = man_armor.802.d

	is_triggered_only = yes

	option = {
		name = man_armor.802.a
		trigger = { tag = MAN }
	}

	option = {
		name = man_armor.802.b
		trigger = {
			tag = SOV 
		}
		MAN = { army_experience = 25 }
		MAN = { economy_fatigue_level_down_1 = yes }
		if = {
			limit = {
				has_global_flag = siberia_secured 
			}
			563 = {
				damage_building = {
					type = infrastructure
					damage = 1
				}
				damage_building = {
					type = industrial_complex
					damage = 1
				}
			}
		}
		else = {
			economy_fatigue_level_up_1 = yes
			add_stability = -0.01
			563 = {
				damage_building = {
					type = infrastructure
					damage = 1
				}
				damage_building = {
					type = industrial_complex
					damage = 1
				}
				damage_building = {
					type = arms_factory
					damage = 1
				}
			}
		}
	}
}
